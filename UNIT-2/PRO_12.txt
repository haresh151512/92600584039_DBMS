-- EX-12
-- WRITE A PROCEDURE THAT SEARCH WHETHER THE GIVEN EMPLOYEE NUMBER
-- IS PRESENT OR NOT IN THE TABLE.
-- IF PRESENT THEN PRINT NAME, SALARY AND DEPARTMENT NUMBER.
-- USE BOTH IN AND OUT MODE VARIABLES.
-- WRITE A PL/SQL BLOCK TO CALL THE SEARCH_EMP PROCEDURE.

SET SERVEROUTPUT ON;

CREATE OR REPLACE PROCEDURE SEARCH_EMP(
    P_EMP_ID   IN EMP.EID%TYPE,
    P_EMP_NAME OUT EMP.ENAME%TYPE,
    P_EMP_SAL  OUT EMP.BASICSAL%TYPE,
    P_EMP_DNO  OUT EMP.DEPTNO%TYPE
)
AS
BEGIN

    SELECT ENAME, BASICSAL, DEPTNO
    INTO P_EMP_NAME, P_EMP_SAL, P_EMP_DNO
    FROM EMP
    WHERE EID = P_EMP_ID;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('EMPLOYEE ID ' || P_EMP_ID || ' NOT FOUND');

END;
/

DECLARE
    G_NAME EMP.ENAME%TYPE;
    G_SAL  EMP.BASICSAL%TYPE;
    G_DNO  EMP.DEPTNO%TYPE;
BEGIN

    SEARCH_EMP(102, G_NAME, G_SAL, G_DNO);
        DBMS_OUTPUT.PUT_LINE('EMPLOYEE NAME: ' || G_NAME);
        DBMS_OUTPUT.PUT_LINE('BASIC SALARY: ' || G_SAL);
        DBMS_OUTPUT.PUT_LINE('DEPARTMENT NO: ' || G_DNO);
END;
/
