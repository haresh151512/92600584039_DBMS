/*
Write a function that returns the age category of an employee.
Categories: 
Age < 18 -> Minor
18 to 60 -> Adult
Age > 60 -> Senior
*/


CREATE OR REPLACE FUNCTION GET_AGE_CATEGORY (
    P_AGE IN NUMBER
) RETURN VARCHAR2
IS
    V_CATEGORY VARCHAR2(20);
BEGIN
    IF P_AGE IS NULL OR P_AGE < 0 THEN
        RAISE_APPLICATION_ERROR(-20001, 'INVALID AGE.');
    END IF;

    IF P_AGE < 18 THEN
        V_CATEGORY := 'MINOR';
    ELSIF P_AGE BETWEEN 18 AND 59 THEN
        V_CATEGORY := 'ADULT';
    ELSE
        V_CATEGORY := 'SENIOR';
    END IF;

    RETURN V_CATEGORY;
END;
/


BEGIN
    DBMS_OUTPUT.PUT_LINE(GET_AGE_CATEGORY(15));
    DBMS_OUTPUT.PUT_LINE(GET_AGE_CATEGORY(30));
    DBMS_OUTPUT.PUT_LINE(GET_AGE_CATEGORY(65));
END;
/
