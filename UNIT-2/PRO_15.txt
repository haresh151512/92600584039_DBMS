/*
Write a PACKAGE which includes Package Declaration, Package
Body and call the individual object from the Package.
*/

-- 1. PACKAGE SPECIFICATION (DECLARATION)
CREATE OR REPLACE PACKAGE MATH_OPERATIONS AS
    PROCEDURE ADD_NUMBERS(P_NUM1 IN NUMBER, P_NUM2 IN NUMBER);
    FUNCTION MULTIPLY_NUMBERS(P_NUM1 IN NUMBER, P_NUM2 IN NUMBER) RETURN NUMBER;
    G_LAST_RESULT NUMBER;
END MATH_OPERATIONS;
/
------------------------------------------------------------

-- 2. PACKAGE BODY (IMPLEMENTATION)
CREATE OR REPLACE PACKAGE BODY MATH_OPERATIONS AS
    PROCEDURE ADD_NUMBERS(P_NUM1 IN NUMBER, P_NUM2 IN NUMBER) IS
    BEGIN
        G_LAST_RESULT := P_NUM1 + P_NUM2;
        DBMS_OUTPUT.PUT_LINE('SUM: ' || G_LAST_RESULT);
    END ADD_NUMBERS;
    FUNCTION MULTIPLY_NUMBERS(P_NUM1 IN NUMBER, P_NUM2 IN NUMBER) RETURN NUMBER IS
    BEGIN
        G_LAST_RESULT := P_NUM1 * P_NUM2;
        RETURN G_LAST_RESULT;
    END MULTIPLY_NUMBERS;

END MATH_OPERATIONS;
/
------------------------------------------------------------

-- 3. CALLING PACKAGE OBJECTS
SET SERVEROUTPUT ON;

BEGIN
    MATH_OPERATIONS.ADD_NUMBERS(10, 20);
    DBMS_OUTPUT.PUT_LINE('PRODUCT: ' || MATH_OPERATIONS.MULTIPLY_NUMBERS(5, 6));
    DBMS_OUTPUT.PUT_LINE('LAST RESULT STORED IN G_LAST_RESULT: ' || MATH_OPERATIONS.G_LAST_RESULT);
END;
/
