/*
--EX-13.WRITE A PROCEDURE THAT ACCEPT EMPLOYEE ID AND REMOVE RECORDS FROM
	EMPLOYEE TABLE. ALSO, CREATE A CALLING PROGRAM TO CALL PROCEDURE.
*/

CREATE OR REPLACE PROCEDURE DELETE_EMP(
EMP_ID IN EMP.EID%TYPE
)AS

BEGIN
    DELETE FROM EMP
    WHERE EID=EMP_ID;

EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE(TO_CHAR(EMP_ID) ||' NO DATA FOND ');
END;
/  


-------------------------
SET SERVEROUTPUT ON;

DECLARE
    EID EMP.EID%TYPE;
BEGIN
    EID := &EMPLOYEE_ID;
    DELETE_EMP(EID);
END;
/
