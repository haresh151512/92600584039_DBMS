-----WAP TO PRINT VALUES 1 TO 100 NUMBERS USING FOR LOOP-----
SET SERVEROUTPUT ON;
DECLARE
BEGIN
	FOR I IN 1..100
		LOOP
			DBMS_OUTPUT.PUT_LINE(I);
		END LOOP;
END;
/


-----WAP TO THAT PRINT VALUES 1 TO 100 NUBERS USING LOOP COMMAND-----
SET SERVEROUTPUT ON;
DECLARE
	COUNTER NUMBER := 1;
BEGIN
	LOOP
		DBMS_OUTPUT.PUT_LINE(COUNTER);
		COUNTER := COUNTER + 1;
		EXIT WHEN COUNTER > 100;
		END LOOP;
END;
/

-----WAP THAT PRINT VALUES 1 TO 100 NUMBERS USING WHILE LOOP COMMAND-----
SET SERVEROUTPUT ON;
DECLARE
	COUNTER NUMBER := 1;
BEGIN
	WHILE COUNTER <=100
		LOOP
			DBMS_OUTPUT.PUT_LINE(COUNTER);
			COUNTER := COUNTER + 1;
		END LOOP;
END;
/
