--EX-2.WRITE A PL/SQL BLOCK TO ACCEPT PRODUCT NAME, QTY AND PRICE FROM USER AND THEN CALCULATE DISCOUNT IN RS. BASED ON THE GIVEN (%).

SET SERVEROUTPUT ON;

DECLARE
    P_NAME      VARCHAR2(20) := '';
    QTY         REAL;
    PRICE       REAL;
    DISCOUNT    REAL;
    TOTAL       REAL;
    DISCOUNT_RS REAL;
    NET_AMOUNT  REAL;
BEGIN
    P_NAME:='&ENTER_PRODUCT_NAME';
    QTY:=&ENTER_QUANTITY;
    PRICE:=&ENTER_PRICE;
    DISCOUNT:=&ENTER_DISCOUNT_PERCENTAGE;
    
   
    TOTAL := QTY * PRICE;
    DISCOUNT_RS := (TOTAL * DISCOUNT) / 100;
    NET_AMOUNT := TOTAL - DISCOUNT_RS;

    DBMS_OUTPUT.PUT_LINE('PRODUCT NAME       : ' || P_NAME);
    DBMS_OUTPUT.PUT_LINE('QUANTITY           : ' || QTY);
    DBMS_OUTPUT.PUT_LINE('PRICE PER UNIT     : ' || PRICE);
    DBMS_OUTPUT.PUT_LINE('TOTAL AMOUNT       : ' || TOTAL);
    DBMS_OUTPUT.PUT_LINE('DISCOUNT (%)       : ' || DISCOUNT);
    DBMS_OUTPUT.PUT_LINE('DISCOUNT (RS.)     : ' || DISCOUNT_RS);
    DBMS_OUTPUT.PUT_LINE('NET AMOUNT         : ' || NET_AMOUNT);
END;
