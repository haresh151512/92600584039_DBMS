--WAP TO CALCULATE SALARY--
CREATE TABLE EMPLOYEE
(
	EID NUMBER(5) PRIMARY KEY,
	ENAME VARCHAR(50),
	DEPTNO NUMBER(5),
	DEPTNAME VARCHAR(50),
	GENDER VARCHAR(50),
	AGE NUMBER(5),
	BASICSAL REAL
);

SET SERVEROUTPUT ON;
DECLARE
	Eid EMPLOYEE.EID%TYPE;
	EBASICSAL EMPLOYEE.BASICSAL%TYPE;
	GROSSSALARY REAL;
	DA REAL;
	HRA REAL;
	PF REAL;
	MEDICAL REAL;
BEGIN
 	Eid :=&EID;
	SELECT BASICSAL INTO EBASICSAL
	FROM EMPLOYEE
	WHERE EID=Eid;

	DA := EBASICSAL*(50/100);
	HRA := EBASICSAL*(15/100);
	MEDICAL :=500;
	PF :=EBASICSAL*(10/100);
	GROSSSALARY := EBASICSAL + (DA + HRA + MEDICAL) - PF;

	DBMS_OUTPUT.PUT_LINE('GROSS SALARY : ' || GROSSSALARY);
END;
