--EX-8.WRITE A PL/SQL BLOCK TO DELETE THE RECORD OF EMPLOYEE FOR GIVEN EID.

SET SERVEROUTPUT ON;

DECLARE
    EID_NO EMP.EID%TYPE;

BEGIN
    EID_NO := &ENTER_EID;

    WHERE EID = EID_NO;

    IF SQL%ROWCOUNT > 0 THEN
        DBMS_OUTPUT.PUT_LINE('EMPLOYEE RECORD DELETED SUCCESSFULLY.');
    ELSE
        DBMS_OUTPUT.PUT_LINE('EMPLOYEE ID NOT FOUND.');
    END IF;

END;
