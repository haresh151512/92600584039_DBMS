--EX-7.WRITE A PL/SQL BLOCK WHICH CONVERTS TEMPERATURE FROM CELSIUS TO FAHRENHEIT.

SET SERVEROUTPUT ON;

DECLARE
    CELSIUS    NUMBER(5);
    FAHRENHEIT NUMBER(5);

BEGIN
    CELSIUS := &ENTER_CELSIUS;
    FAHRENHEIT := (CELSIUS * 9/5) + 32;

    DBMS_OUTPUT.PUT_LINE('---------------------------');
    DBMS_OUTPUT.PUT_LINE('TEMPERATURE IN CELSIUS    : ' || CELSIUS || ' C');
    DBMS_OUTPUT.PUT_LINE('TEMPERATURE IN FAHRENHEIT : ' || FAHRENHEIT || ' F');
    DBMS_OUTPUT.PUT_LINE('---------------------------');
END;
