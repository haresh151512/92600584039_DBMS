/*
---EX : 9 CREATE A PACKAGE NAMED PKG_EMPLOYEE CONTAINING:
	FUNCTION TO CALCULATE THE ANNUAL SALARY OF AN EMPLOYEE.
	PROCEDURE TO SHOW EMPLOYEE INFORMATION USING THE EMPLOYEE ID.
	CALL EACH PACKAGE OBJECT SEPARATELY.
*/

SET SERVEROUTPUT ON;

CREATE OR REPLACE PACKAGE PKG_EMPLOYEE
AS
    FUNCTION ANNUAL_SALARY(
        P_EMP_ID IN EMP.EID%TYPE
    )
    RETURN NUMBER;

    PROCEDURE SHOW_EMPLOYEE(
        P_EMP_ID IN EMP.EID%TYPE
    );

END PKG_EMPLOYEE;
/


CREATE OR REPLACE PACKAGE BODY PKG_EMPLOYEE
AS

    FUNCTION ANNUAL_SALARY(
        P_EMP_ID IN EMP.EID%TYPE
    )
    RETURN NUMBER
    IS
        V_SALARY EMP.BASICSAL%TYPE;
    BEGIN

        SELECT BASICSAL
        INTO V_SALARY
        FROM EMP
        WHERE EID = P_EMP_ID;

        RETURN V_SALARY * 12;

    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            DBMS_OUTPUT.PUT_LINE('EMPLOYEE ID NOT FOUND');
            RETURN 0;
    END ANNUAL_SALARY;

    PROCEDURE SHOW_EMPLOYEE(
        P_EMP_ID IN EMP.EID%TYPE
    )
    IS
        V_NAME EMP.ENAME%TYPE;
        V_SAL  EMP.BASICSAL%TYPE;
        V_DNO  EMP.DEPTNO%TYPE;
    BEGIN

        SELECT ENAME, BASICSAL, DEPTNO
        INTO V_NAME, V_SAL, V_DNO
        FROM EMP
        WHERE EID = P_EMP_ID;

        DBMS_OUTPUT.PUT_LINE('EMPLOYEE ID   : ' || P_EMP_ID);
        DBMS_OUTPUT.PUT_LINE('EMPLOYEE NAME : ' || V_NAME);
        DBMS_OUTPUT.PUT_LINE('SALARY        : ' || V_SAL);
        DBMS_OUTPUT.PUT_LINE('DEPARTMENT NO : ' || V_DNO);

    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            DBMS_OUTPUT.PUT_LINE('EMPLOYEE ID NOT FOUND');
    END SHOW_EMPLOYEE;

END PKG_EMPLOYEE;
/


SELECT PKG_EMPLOYEE.ANNUAL_SALARY(102) AS ANNUAL_SALARY
FROM DUAL;


BEGIN
    PKG_EMPLOYEE.SHOW_EMPLOYEE(102);
END;
/
