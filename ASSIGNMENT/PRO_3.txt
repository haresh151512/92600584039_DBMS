--WRITE A PL/SQL PROGRAM TO GANERATE THE MULTIPLICATION TABLE OF A USER-ENTERED NUMBER USING: FOR LOOP AND WHILE LOOP

SET SERVEROUTPUT ON;
DECLARE
	VNUM	NUMBER := &NUMBER; 
    	VCOUNT	NUMBER := 1;
                  
BEGIN
	DBMS_OUTPUT.PUT_LINE('FOR LOOP TABLE');
	FOR i IN 1..10 LOOP
        	DBMS_OUTPUT.PUT_LINE(VNUM || ' x ' || i || ' = ' || (VNUM * i));
    	END LOOP;
	DBMS_OUTPUT.PUT_LINE(CHR(10));
	DBMS_OUTPUT.PUT_LINE('WHILE LOOP TABLE');
    	WHILE VCOUNT <= 10
	LOOP
        	DBMS_OUTPUT.PUT_LINE(VNUM || ' x ' || VCOUNT || ' = ' || (VNUM * VCOUNT));
       VCOUNT := VCOUNT + 1;    
	END LOOP;
END;
/
