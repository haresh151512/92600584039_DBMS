/*
Create a procedure that accepts Department Number and displays:
Total Employees
Average Salary
Maximum Salary
Use OUT parameters.
*/


CREATE OR REPLACE PROCEDURE DISPLAY(
    DPNO IN EMP.DEPTNO%TYPE,
    TOTAL_EMP OUT NUMBER,
    AVG_SALARY OUT NUMBER,
    MAX_SALARY OUT NUMBER
) AS
BEGIN
    SELECT COUNT(*),AVG(BASICSAL),MAX(BASICSAL)
    INTO   TOTAL_EMP,
           AVG_SALARY,
           MAX_SALARY
    FROM   EMP
    WHERE  DEPTNO=DPNO;
EXCEPTION
	WHERE OTHER THEN
		DBMS_OUTPUT.PUT_LINE('NOT FOUND');
END;
/	


VAR @TOTAL_EMP
VAR @AVG_SALARY
VAR @MAX_SALARY

EXECUTE DISPLAY(101,@TOTAL_EMP,@AVG_SALARY,@MAX_SALARY)
