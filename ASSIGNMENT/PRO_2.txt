--WAP TO CALCULATE GRADE--
CREATE TABLE RESULT151512
(
	ROLLNO NUMBER(5) PRIMARY KEY,
	SNAME VARCHAR2(50),
	SUB1 NUMBER(3) CHECK (SUB1 BETWEEN 0 AND 100),
	SUB2 NUMBER(3) CHECK (SUB2 BETWEEN 0 AND 100),
	SUB3 NUMBER(3) CHECK (SUB3 BETWEEN 0 AND 100),
	SUB4 NUMBER(3) CHECK (SUB4 BETWEEN 0 AND 100),
	SUB5 NUMBER(3) CHECK (SUB5 BETWEEN 0 AND 100),
	TOTAL NUMBER(5),
	PER NUMBER(5,2),
	GRADE VARCHAR2(2)
);

INSERT INTO RESULT151512(ROLLNO,SNAME,SUB1,SUB2,SUB3,SUB4,SUB5)
VALUES(1,'HARESH',75,80,85,65,70);

SET SERVEROUTPUT ON;
DECLARE
	ROLLNO RESULT151512.ROLLNO%TYPE;
	SNAME RESULT151512.SNAME%TYPE;
	SUB1 RESULT151512.SUB1%TYPE;
	SUB2 RESULT151512.SUB2%TYPE;
	SUB3 RESULT151512.SUB3%TYPE;
	SUB4 RESULT151512.SUB4%TYPE;
	SUB5 RESULT151512.SUB5%TYPE;
	TOTAL RESULT151512.TOTAL%TYPE;
	PER RESULT151512.PER%TYPE;
	GRADE RESULT151512.GRADE%TYPE;
BEGIN
	ROLLNO := &ENTER_RNO_NO;
	SELECT SNAME,SUB1,SUB2,SUB3,SUB4,SUB5
	INTO SNAME,SUB1,SUB2,SUB3,SUB4,SUB5
	FROM RESULT151512
	WHERE ROLLNO = ROLLNO;
	TOTAL := SUB1+SUB2+SUB3+SUB4+SUB5;
	PER := TOTAL/5;

	IF SUB1<35 OR SUB2<35 OR SUB3<35 OR SUB4<35 OR SUB5<35 THEN
		GRADE := 'F' ;
		DBMS_OUTPUT.PUT_LINE('RESULT : FAIL');
	ELSE
		IF PER >= 90 THEN
			GRADE := 'A+';
		ELSIF PER >= 80 THEN
			GRADE := 'A';
		ELSIF PER >= 70 THEN
			GRADE := 'B';
		ELSIF PER >= 60 THEN
			GRADE := 'C';
		ELSIF PER >= 50 THEN
			GRADE := 'D';
		ELSIF PER >= 40 THEN
			GRADE := 'E';
		ELSE
			GRADE := 'F';
		END IF;
		DBMS_OUTPUT.PUT_LINE('RESULT : PASS');
   	 END IF;	
	DBMS_OUTPUT.PUT_LINE('--------------------');
	DBMS_OUTPUT.PUT_LINE('ROLL NO         : ' || ROLLNO);
	DBMS_OUTPUT.PUT_LINE('NAME              : ' || SNAME);
	DBMS_OUTPUT.PUT_LINE('PERCENTAGE : ' || PER || '%');
	DBMS_OUTPUT.PUT_LINE('GRADE            : ' || GRADE);
	DBMS_OUTPUT.PUT_LINE('--------------------');
END;
