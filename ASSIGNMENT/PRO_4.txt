--WRITE PL/SQL BLOCK TO ACCEPT AN EMPLOYEE ID AND: RETRIVE EMPLOYEE DETAILS , CALCULATE YEARLY SALARY , DISPLAY WHETHER EMPLOYEE BELONGS TO HIGH INCOME (>₹6,00,000/YEAR) OR NORMAL INCOME.

CREATE TABLE EMPLOYEES
(
	EID NUMBER(5) PRIMARY KEY,
	FIRST_NAME VARCHAR(50),
	LAST_NAME VARCHAR(50),
	MONTHLY REAL
);

INSERT INTO EMPLOYEES(EID,FIRST_NAME,LAST_NAME,MONTHLY)
VALUES(1,'HARESH','KANANI',50000);

INSERT INTO EMPLOYEES(EID,FIRST_NAME,LAST_NAME,MONTHLY)
VALUES(2,'VISHVRAJ','ZALA',70000);

INSERT INTO EMPLOYEES(EID,FIRST_NAME,LAST_NAME,MONTHLY)
VALUES(3,'DIPEN','GHODASHARA',100000);

DECLARE
   EID EMPLOYEES.EID%TYPE := &EID;
   FIRST_NAME EMPLOYEES.FIRST_NAME%TYPE;
   LAST_NAME  EMPLOYEES.LAST_NAME%TYPE;
   MONTHLY    EMPLOYEES.MONTHLY%TYPE;
   YEARLY     NUMBER;

BEGIN
    SELECT FIRST_NAME, LAST_NAME, MONTHLY
    INTO FIRST_NAME,LAST_NAME, MONTHLY
    FROM EMPLOYEES
    WHERE EID = EID;
    YEARLY := MONTHLY * 12;

    DBMS_OUTPUT.PUT_LINE('Employee ID: ' || EID);
    DBMS_OUTPUT.PUT_LINE('Name: ' ||FIRST_NAME || ' ' ||LAST_NAME);
    DBMS_OUTPUT.PUT_LINE('Monthly Salary: ₹' || MONTHLY);
    DBMS_OUTPUT.PUT_LINE('Yearly Salary: ₹' || YEARLY);

    IF YEARLY > 600000 THEN
        DBMS_OUTPUT.PUT_LINE('Income Status: High Income');
    ELSE
        DBMS_OUTPUT.PUT_LINE('Income Status: Normal Income');
    END IF;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('No EMPLOYEE found with ID ' || EID);
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('An error occurred: ' || SQLERRM);
END;
