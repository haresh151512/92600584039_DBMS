/*
---EX : - 7 CREATE A PROCEDURE THAT ACCEPTS DEPARTMENT NUMBER AND DISPLAYS:
	TOTAL EMPLOYEES
	AVERAGE SALARY
	MAXIMUM SALARY
	USE OUT PARAMETERS.
*/

CREATE TABLE EMPLOYEE (
    EMP_ID NUMBER,
    EMP_NAME VARCHAR2(50),
    SALARY NUMBER
);

SELECT EMP_ID,
       EMP_NAME,
       SALARY,
       ANNUAL_SALARY(SALARY) AS ANNUAL_SALARY
FROM EMPLOYEE;

CREATE OR REPLACE FUNCTION ANNUAL_SALARY(P_SALARY NUMBER)
RETURN NUMBER
IS
BEGIN
    RETURN P_SALARY * 12;
END;
/

SELECT ANNUAL_SALARY(25000) AS ANNUAL_SALARY
FROM DUAL;
