SET SERVEROUTPUT ON;
DECLARE
	NUM1 NUMBER(4); NUM2 NUMBER(4);
	NUM3 NUMBER(4);
	MAXNUM NUMBER(4);
	TOTAL NUMBER(4);
BEGIN
	NUM1:=&NUM1;
	NUM2:=&NUM2;
	NUM3:=&NUM3; MAXNUM :=NUM1;
	TOTAL:=NUM1+NUM2+NUM3;
	IF NUM1=NUM2 AND NUM2=NUM3 THEN
	DBMS_OUTPUT.PUT_LINE('ALL NUMBER IS EQUAL');
	ELSE
		IF NUM2 > MAXNUM THEN
			MAXNUM := NUM2;
		END IF;
		IF NUM3 > MAXNUM THEN
			MAXNUM := NUM3;
		END IF;
		DBMS_OUTPUT.PUT_LINE('MAX NUMBER IS : ' || MAXNUM);
	END IF;
	DBMS_OUTPUT.PUT_LINE('AVERAGE OF 3 NUMBERS '||TOTAL/3);
END;
/
